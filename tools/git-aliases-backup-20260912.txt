# 原 git alias 定义（pushall / pushgitee / pushgithub / st）

alias.pushall !f() { G="/c/Program Files/Git/mingw64/bin/git.exe"; echo "== Gitee =="; "$G" push gitee main:main && "$G" push gitee main:master; echo "== GitHub =="; L=$("$G" rev-parse HEAD); if "$G" push github-ssh main:refs/heads/main >/dev/null 2>&1; then echo "  (SSH 通道)"; else echo "  SSH 不通，回退 HTTPS"; for i in 1 2 3 4 5; do "$G" -c http.proxy= -c http.lowSpeedLimit=0 -c http.lowSpeedTime=999999 push origin main:refs/heads/main >/dev/null 2>&1 && break; sleep 3; done; fi; R=$("$G" ls-remote github-ssh refs/heads/main 2>/dev/null | cut -f1); [ -z "$R" ] && R=$("$G" -c http.proxy= ls-remote origin refs/heads/main 2>/dev/null | cut -f1); if [ "$R" = "$L" ]; then echo "  GitHub 已确认同步: $(echo $R | cut -c1-7)"; else echo "  GitHub 未确认（远程=${R:-无响应} / 本地=$(echo $L | cut -c1-7)），需补推"; fi; }; f
alias.pushgitee !f() { G='/c/Program Files/Git/mingw64/bin/git.exe'; "$G" push gitee main:main && "$G" push gitee main:master; }; f
alias.st status -sb
alias.pushgithub !f() { G="/c/Program Files/Git/mingw64/bin/git.exe"; L=$("$G" rev-parse HEAD); if "$G" push github-ssh main:refs/heads/main >/dev/null 2>&1; then echo "走 SSH 通道"; else echo "SSH 不通，回退 HTTPS"; for i in 1 2 3 4 5 6; do "$G" -c http.proxy= -c http.lowSpeedLimit=0 -c http.lowSpeedTime=999999 push origin main:refs/heads/main >/dev/null 2>&1 && break; sleep 3; done; fi; R=$("$G" ls-remote github-ssh refs/heads/main 2>/dev/null | cut -f1); [ -z "$R" ] && R=$("$G" -c http.proxy= ls-remote origin refs/heads/main 2>/dev/null | cut -f1); if [ "$R" = "$L" ]; then echo "GitHub 已同步: $(echo $R | cut -c1-7)"; else echo "GitHub 未同步（远程=${R:-无响应}）"; fi; }; f
